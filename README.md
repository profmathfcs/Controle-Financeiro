# Q1: f(x) = -2x^2 + 7x - 5
xv1 = -7 / (2 * -2)
yv1 = -2 * (xv1**2) + 7 * xv1 - 5
print(f"Q1: Xv={xv1}, Yv={yv1}")

# Q2: f(x) = 2x^2 + 4x - 10
xv2 = -4 / (2 * 2)
yv2 = 2 * (xv2**2) + 4 * xv2 - 10
print(f"Q2: Xv={xv2}, Yv={yv2}")

# Q3: h(x) = -x^2 / 2 + 6x - 10
xv3 = -6 / (2 * -0.5)
yv3 = -0.5 * (xv3**2) + 6 * xv3 - 10
print(f"Q3: Xv={xv3}, Yv={yv3}")

# Q4: p(t) = -2t^2 + 33t + 30
tv4 = -33 / (2 * -2)
print(f"Q4: tv={tv4}")

# Q5: T(t) = -t^2 + 24t - 100
tv5 = -24 / (2 * -1)
Tv5 = -(tv5**2) + 24 * tv5 - 100
print(f"Q5: tv={tv5}, Tv={Tv5}")

# Q6: y = -2x^2 + 1000x
xv6 = -1000 / (2 * -2)
print(f"Q6: Xv={xv6}")

# Q7: L(v) = -v^2 + 7v - 6
vv7 = -7 / (2 * -1)
Lv7 = -(vv7**2) + 7 * vv7 - 6
print(f"Q7: vv={vv7}, Lv={Lv7}")
